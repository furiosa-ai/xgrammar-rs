4pub enum XGrammarErr {

7    #[error("failed to parse tokenizer config: {0}")]
8    TokenizerParseFailed(String),
9    #[error("invalid tokenizer config: {0}")]
10    InvalidTokenizerConfig(String),
11    #[cfg(feature = "hf_hub")]
12    #[error("failed to download from Hugging Face Hub: {0}")]
13    HuggingFaceDownloadFailed(#[from] crate::huggingface_hub::ApiError),
14    #[error("failed to parse JSON: {0}")]
15    JsonParseFailed(#[from] serde_json::Error),
16    #[error("missing field in JSON: {0}")]
17    MissingJsonField(String),
18    #[error("invalid vocab: {0}")]
19    InvalidVocab(String),
20    // -----------------------------------------------------------------------
21    // Every variant below is converted directly from an error raised by the
22    // underlying xgrammar C++ library. As a binding, we preserve upstream
23    // behavior and pass the message through verbatim, without adding a
24    // display prefix: upstream messages are already self-describing (e.g.
25    // "EBNF lexer error at ...", "Invalid JSON error: ...").
26    // -----------------------------------------------------------------------
27    /// An untyped xgrammar error raised while constructing a `Grammar`.
28    #[error("{0}")]
29    InvalidGrammar(String),
30    /// An untyped xgrammar error raised while compiling a grammar.
31    #[error("{0}")]
32    CompilationError(String),
33    /// An untyped xgrammar error raised by a matcher operation.
34    #[error("{0}")]
35    MatcherError(String),
36    /// An untyped xgrammar error raised while deserializing a
37    /// `TokenizerInfo`.
38    #[error("{0}")]
39    TokenizerInfoError(String),
40    // The five variants below correspond one-to-one to the upstream
41    // `xgrammar::XGrammarError` subclasses
42    // (thirdparty/xgrammar/include/xgrammar/exception.h), mapped via
43    // `XGrammarError::GetType()`.
44    /// The input JSON text is malformed.
45    #[error("{0}")]
46    InvalidJson(String),
47    /// The JSON schema is invalid or unsatisfiable (defined upstream but not
48    /// constructed as of xgrammar v0.2.3 — reserved for forward
49    /// compatibility).
50    #[error("{0}")]
51    InvalidJsonSchema(String),
52    /// The structural tag specification is invalid.
53    #[error("{0}")]
54    InvalidStructuralTag(String),
55    /// The serialized data was produced by an incompatible xgrammar
56    /// serialization version.
57    #[error("{0}")]
58    DeserializeVersion(String),
59    /// The serialized data does not follow the expected format.
60    #[error("{0}")]
61    DeserializeFormat(String),
62}
