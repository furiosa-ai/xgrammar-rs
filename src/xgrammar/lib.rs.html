98    pub fn from_backend_str(
99        backend_str: &str,
100        vocab_size: Option<usize>,
101        stop_token_ids: Vec<TokenId>,
102    ) -> self::Result<Self> {
103        let tokenizer = tokenizers::Tokenizer::from_str(backend_str).map_err(|e| {
104            XGrammarErr::TokenizerParseFailed(format!("failed to parse tokenizer: {}", e))
105        })?;
106        let vocab_map = tokenizer.get_vocab(true); // with added special tokens
107        let max_id = vocab_map
108            .values()
109            .max()
110            .ok_or(XGrammarErr::InvalidTokenizerConfig("Vocab map is empty".to_string()))?;
111        let tokenizer_vocab_size = std::cmp::max(vocab_map.len(), (max_id + 1) as usize);
112        if let Some(vocab_size) = vocab_size {
113            if vocab_size != tokenizer_vocab_size {
114                tracing::warn!(
115                    "Provided vocab_size {} does not match tokenizer vocab size {}. Using provided vocab_size.",
116                    vocab_size,
117                    tokenizer_vocab_size
118                );
119            }
120        }
121        let final_vocab_size = vocab_size.unwrap_or(tokenizer_vocab_size);
122        let tokenizer_metadata = Self::detect_metadata_from_hf(backend_str);
123        let vocab_type = tokenizer_metadata.vocab_type;
124        let add_prefix_space = tokenizer_metadata.add_prefix_space;
125
126        Self::new(vocab_map, vocab_type, final_vocab_size, stop_token_ids, add_prefix_space)
127    }

129    pub fn parse_eos_token(path: &Path, json_key: &str) -> Option<Vec<i32>> {
130        let contents = std::fs::read_to_string(path).ok()?;
131        let json: Value = serde_json::from_str(&contents).ok()?;
132        match json.get(json_key) {
133            Some(Value::Number(num)) if num.is_i64() => Some(vec![num.as_i64().unwrap() as i32]),
134            Some(Value::Array(arr)) => {
135                let mut eos_tokens = Vec::new();
136                for item in arr {
137                    if let Value::Number(num) = item {
138                        if num.is_i64() {
139                            eos_tokens.push(num.as_i64().unwrap() as i32);
140                        } else {
141                            return None;
142                        }
143                    } else {
144                        return None;
145                    }
146                }
147                Some(eos_tokens)
148            }
149            _ => None,
150        }
151    }

153    pub fn from_path<P>(
154        path: P,
155        vocab_size: Option<usize>,
156        stop_token_ids: Option<Vec<TokenId>>,
157    ) -> Result<Self>
158    where
159        P: AsRef<Path>,
160    {
161        let path = path.as_ref();
162        let tokenizer_json_path = path.join(TOKENIZER_FILE);
163        let backend_str = std::fs::read_to_string(&tokenizer_json_path)
164            .map_err(XGrammarErr::TokenizerLoadFailed)?;
165
166        let eos_token = Self::parse_eos_token(&path.join(GENERATION_CONFIG_FILE), EOS_TOKEN_ID_KEY)
167            .or_else(|| Self::parse_eos_token(&path.join(HF_CONFIG_FILE), EOS_TOKEN_ID_KEY));
168
169        let mut stop_token_ids = stop_token_ids.unwrap_or_default();
170        stop_token_ids.extend(eos_token.unwrap_or_default());
171        stop_token_ids.dedup();
172
173        Self::from_backend_str(&backend_str, vocab_size, stop_token_ids)
174    }

177    pub fn from_pretrained(
178        tokenizer_id: &str,
179        revision: Option<String>,
180        vocab_size: Option<usize>,
181        stop_token_ids: Option<Vec<i32>>,
182    ) -> Result<TokenizerInfo> {
183        use huggingface_hub::{Params, Repo, RepoType, compile_glob_pattern, snapshot_download};
184
185        let allow_patterns = compile_glob_pattern(TOKENIZER_ALLOW_PATTERN).map_err(|e| {
186            XGrammarErr::TokenizerParseFailed(format!("Failed to compile glob patterns: {}", e))
187        })?;
188        let download_options =
189            Some(Params { allow_patterns: Some(allow_patterns), ..Default::default() });
190
191        let repo = Repo::with_revision(
192            tokenizer_id.to_string(),
193            RepoType::Model,
194            revision.unwrap_or("main".to_string()),
195        );
196        let tokenizer_dir = snapshot_download(repo, download_options)?;
197        Self::from_path(tokenizer_dir, vocab_size, stop_token_ids)
198    }

200    fn new(
201        vocab_map: HashMap<String, u32>,
202        vocab_type: VocabType,
203        vocab_size: usize,
204        stop_token_ids: Vec<i32>,
205        add_prefix_space: bool,
206    ) -> self::Result<Self> {
207        // Ensure the vocab size is at least as large as the max id in the vocab map
208        let mut encoded_vocab = vec![CString::new("").unwrap(); vocab_size];
209
210        // Fill the encoded_vocab with tokens from the vocab_map
211        for (token, idx) in vocab_map.iter() {
212            assert!(
213                (*idx as usize) < vocab_size,
214                "Token ID {} exceeds vocab size {}",
215                idx,
216                vocab_size
217            );
218            encoded_vocab[*idx as usize] =
219                CString::new(token.as_str()).expect("fail to convert a token to CString");
220        }
221
222        let encoded_vocab_ptr: Vec<_> = encoded_vocab.iter().map(|s| s.as_ptr()).collect();
223        let encoded_vocab_ptr_ptr = encoded_vocab_ptr.as_ptr();
224        let vocab_size_i32 = vocab_size as i32;
225        let stop_token_ids_ptr = stop_token_ids.as_ptr();
226        let stop_token_ids_len = stop_token_ids.len();
227
228        Ok(cpp!(unsafe [
229            encoded_vocab_ptr_ptr as "const char* const*",
230            vocab_type as "xgrammar::VocabType",
231            vocab_size_i32 as "int",
232            stop_token_ids_ptr as "const int32_t*",
233            stop_token_ids_len as "size_t",
234            add_prefix_space as "bool"
235        ] -> TokenizerInfo as "xgrammar::TokenizerInfo" {
236            std::vector<std::string> encoded_vocab;
237            for (int i = 0; i < vocab_size_i32; ++i) {
238                encoded_vocab.push_back(std::string(encoded_vocab_ptr_ptr[i]));
239            }
240            std::vector<int32_t> stop_token_ids(stop_token_ids_ptr, stop_token_ids_ptr + stop_token_ids_len);
241
242            return xgrammar::TokenizerInfo(
243                encoded_vocab,
244                vocab_type,
245                vocab_size_i32,
246                stop_token_ids,
247                add_prefix_space
248            );
249        }))
250    }

280    fn detect_metadata_from_hf(backend_str: &str) -> MetadataFromHF {
281        let backend_str =
282            CString::new(backend_str).expect("Failed to convert backend_str to CString");
283        let backend_str_ptr = backend_str.as_ptr();
284
285        cpp!(unsafe [backend_str_ptr as "const char*"] -> MetadataFromHF as "MetadataFromHF" {
286            const std::string &backend_str(backend_str_ptr);
287            std::string metadata_str = TokenizerInfo::DetectMetadataFromHF(backend_str);
288            picojson::value v;
289            std::string err = picojson::parse(v, metadata_str);
290            if (!err.empty()) {
291                throw std::runtime_error("Failed to parse metadata: " + err);
292            }
293            const picojson::object& metadata = v.get<picojson::object>();
294
295            MetadataFromHF metadata_from_hf;
296            metadata_from_hf.vocab_type = static_cast<xgrammar::VocabType>(metadata["vocab_type"].get<double>());
297            metadata_from_hf.add_prefix_space = metadata["add_prefix_space"].get<bool>();
298            return metadata_from_hf;
299        })
300    }

415    /// Compile a grammar from a JSON schema string.
416    ///
417    /// This method compiles a JSON schema specification into a grammar that enforces
418    /// the schema constraints during text generation.
419    ///
420    /// # Arguments
421    /// * `schema` - The JSON schema string to compile
422    /// * `any_whitespace` - Whether to allow flexible whitespace in the JSON output. None uses true
423    /// * `indent` - Number of spaces for indentation. None means no indentation
424    /// * `separators` - Custom separators as (object_separator, array_separator), e.g., (":", ","). None uses default separators
425    /// * `strict_mode` - Whether to enforce strict JSON schema validation. None uses true
426    /// * `max_whitespace_cnt` - Maximum number of consecutive whitespace characters allowed. None means no limit
427    ///
428    /// # Returns
429    /// * A compiled grammar that can be used with GrammarMatcher
430    pub fn compile_json_schema(
431        &self,
432        schema: &str,
433        any_whitespace: Option<bool>,
434        indent: Option<i32>,
435        separators: Option<(String, String)>,
436        strict_mode: Option<bool>,
437        max_whitespace_cnt: Option<i32>,
438    ) -> CompiledGrammar {
439        let schema_cstring = CString::new(schema).expect("Failed to convert schema to CString");
440        let schema_ptr = schema_cstring.as_ptr();
441        let any_whitespace = any_whitespace.unwrap_or(true);
442        let strict_mode = strict_mode.unwrap_or(true);
443        let has_indent = indent.is_some();
444        let indent_value = indent.unwrap_or(0);
445        let has_separators = separators.is_some();
446        let has_max_whitespace_cnt = max_whitespace_cnt.is_some();
447        let max_whitespace_cnt_value = max_whitespace_cnt.unwrap_or(0);
448
449        let (_obj_sep_cstring, _array_sep_cstring, obj_sep_ptr, array_sep_ptr) =
450            if let Some((obj_sep, array_sep)) = separators {
451                let obj_sep_cstring =
452                    CString::new(obj_sep).expect("Failed to convert object separator to CString");
453                let array_sep_cstring =
454                    CString::new(array_sep).expect("Failed to convert array separator to CString");
455                let obj_sep_ptr = obj_sep_cstring.as_ptr();
456                let array_sep_ptr = array_sep_cstring.as_ptr();
457                (Some(obj_sep_cstring), Some(array_sep_cstring), obj_sep_ptr, array_sep_ptr)
458            } else {
459                (None, None, std::ptr::null(), std::ptr::null())
460            };
461
462        cpp!(unsafe [
463            self as "xgrammar::GrammarCompiler*",
464            schema_ptr as "const char*",
465            any_whitespace as "bool",
466            has_indent as "bool",
467            indent_value as "int",
468            has_separators as "bool",
469            obj_sep_ptr as "const char*",
470            array_sep_ptr as "const char*",
471            strict_mode as "bool",
472            has_max_whitespace_cnt as "bool",
473            max_whitespace_cnt_value as "int"
474        ] -> CompiledGrammar as "xgrammar::CompiledGrammar" {
475            std::string schema_str(schema_ptr);
476            std::optional<int> opt_indent = has_indent ? std::make_optional(indent_value) : std::nullopt;
477            std::optional<std::pair<std::string, std::string>> opt_separators;
478
479            if (has_separators) {
480                opt_separators = std::make_pair(std::string(obj_sep_ptr), std::string(array_sep_ptr));
481            } else {
482                opt_separators = std::nullopt;
483            }
484
485            std::optional<int> opt_max_whitespace_cnt = has_max_whitespace_cnt ? std::make_optional(max_whitespace_cnt_value) : std::nullopt;
486
487            return self->CompileJSONSchema(schema_str, any_whitespace, opt_indent, opt_separators, strict_mode, opt_max_whitespace_cnt);
488        })
489    }

579/// Represents a context-free grammar for grammar-guided text generation.
580///
581/// The Grammar struct supports Extended Backus-Naur Form (EBNF) grammar specifications
582/// following the GBNF specification from llama.cpp. It provides flexible grammar generation
583/// and manipulation for constrained text generation tasks.
584///
585/// # Construction Methods
586///
587/// Grammar can be constructed from various sources:
588/// - [`Grammar::from_ebnf`]: From EBNF grammar strings
589/// - [`Grammar::from_json_schema`]: From JSON schema specifications
590/// - [`Grammar::from_regex`]: From regular expression patterns
591/// - [`Grammar::from_structural_tag`]: From structural tags with embedded schemas
592/// - [`Grammar::builtin_json_grammar`]: Standard JSON grammar
593///
594/// # Grammar Operations
595///
596/// Multiple grammars can be combined using:
597/// - [`Grammar::union`]: Creates a grammar matching any of the input grammars (equivalent to `|` operator)
598/// - [`Grammar::concat`]: Creates a grammar matching concatenated sequences (equivalent to `+` operator)
599impl Grammar {
600    /// Construct a grammar from an EBNF-formatted string.
601    ///
602    /// This method creates a context-free grammar from an Extended Backus-Naur Form (EBNF)
603    /// specification. The grammar follows the GBNF specification from llama.cpp.
604    ///
605    /// # Arguments
606    /// * `ebnf_string` - The EBNF grammar specification string
607    /// * `root_rule_name` - The name of the root rule to use as the entry point. If None, uses "root"
608    ///
609    /// # Returns
610    /// * A Grammar object constructed from the EBNF specification
611    ///
612    /// # Panics
613    /// * Panics if the EBNF string or root rule name contains null bytes
614    ///
615    /// # Example
616    /// ```
617    /// # use xgrammar::Grammar;
618    /// let ebnf = r#"
619    /// root ::= "Hello, " name "!"
620    /// name ::= [A-Z][a-z]+
621    /// "#;
622    /// let grammar = Grammar::from_ebnf(ebnf, Some("root"));
623    /// assert!(!grammar.is_null());
624    /// ```
625    pub fn from_ebnf(ebnf_string: &str, root_rule_name: Option<&str>) -> Self {
626        let ebnf_string_cstring =
627            CString::new(ebnf_string).expect("Failed to convert ebnf_string to CString");
628        let ebnf_string_ptr = ebnf_string_cstring.as_ptr();
629        let root_rule_name = root_rule_name.unwrap_or("root");
630        let root_rule_name_cstring =
631            CString::new(root_rule_name).expect("Failed to convert root_rule_name to CString");
632        let root_rule_name_ptr = root_rule_name_cstring.as_ptr();
633
634        cpp!(unsafe [
635            ebnf_string_ptr as "const char*",
636            root_rule_name_ptr as "const char*"
637        ] -> Grammar as "xgrammar::Grammar" {
638            return xgrammar::Grammar::FromEBNF(std::string(ebnf_string_ptr), std::string(root_rule_name_ptr));
639        })
640    }

642    /// Construct a grammar from a JSON schema string.
643    ///
644    /// This method creates a grammar from a JSON schema specification that enforces schema
645    /// constraints during text generation. The schema can be in JSON string format or
646    /// represent a Pydantic-style model structure.
647    ///
648    /// # Arguments
649    /// * `schema` - The JSON schema string defining the structure to enforce
650    /// * `any_whitespace` - Whether to allow flexible whitespace in the JSON output. When true,
651    ///   any amount of whitespace is allowed between tokens
652    /// * `indent` - Number of spaces for indentation in the JSON output. When specified,
653    ///   produces formatted JSON with the given indentation level
654    /// * `separators` - Custom separators for JSON formatting as (item_separator, key_separator).
655    ///   For example, `(":", ",")` produces compact JSON. When None, uses standard JSON separators
656    /// * `strict_mode` - Whether to enforce strict JSON schema validation. When true, ensures
657    ///   all schema constraints are strictly enforced
658    /// * `max_whitespace_cnt` - Maximum number of consecutive whitespace characters allowed.
659    ///   Useful for preventing excessive whitespace in generated output
660    /// * `print_converted_ebnf` - Whether to print the converted EBNF grammar for debugging purposes
661    ///
662    /// # Returns
663    /// * A Grammar object that enforces the JSON schema constraints
664    ///
665    /// # Panics
666    /// * Panics if the schema string contains null bytes
667    ///
668    /// # Example
669    /// ```
670    /// # use xgrammar::Grammar;
671    /// let schema = r#"{
672    ///   "type": "object",
673    ///   "properties": {
674    ///     "name": {"type": "string"},
675    ///     "age": {"type": "integer"}
676    ///   },
677    ///   "required": ["name", "age"]
678    /// }"#;
679    /// let grammar = Grammar::from_json_schema(
680    ///     schema,
681    ///     Some(true),    // allow flexible whitespace
682    ///     Some(2),       // 2-space indentation
683    ///     None,          // default separators
684    ///     Some(true),    // strict mode
685    ///     None,          // no whitespace limit
686    ///     Some(false)    // don't print EBNF
687    /// );
688    /// assert!(!grammar.is_null());
689    /// ```
690    pub fn from_json_schema(
691        schema: &str,
692        any_whitespace: Option<bool>,
693        indent: Option<i32>,
694        separators: Option<(String, String)>,
695        strict_mode: Option<bool>,
696        max_whitespace_cnt: Option<i32>,
697        print_converted_ebnf: Option<bool>,
698    ) -> Self {
699        let schema_cstring = CString::new(schema).expect("Failed to convert schema to CString");
700        let schema_ptr = schema_cstring.as_ptr();
701        let any_whitespace = any_whitespace.unwrap_or(true);
702        let strict_mode = strict_mode.unwrap_or(true);
703        let print_converted_ebnf = print_converted_ebnf.unwrap_or(false);
704        let has_indent = indent.is_some();
705        let indent_value = indent.unwrap_or(0);
706        let has_separators = separators.is_some();
707        let has_max_whitespace_cnt = max_whitespace_cnt.is_some();
708        let max_whitespace_cnt_value = max_whitespace_cnt.unwrap_or(0);
709
710        let (_obj_sep_cstring, _array_sep_cstring, obj_sep_ptr, array_sep_ptr) =
711            if let Some((obj_sep, array_sep)) = separators {
712                let obj_sep_cstring =
713                    CString::new(obj_sep).expect("Failed to convert object separator to CString");
714                let array_sep_cstring =
715                    CString::new(array_sep).expect("Failed to convert array separator to CString");
716                let obj_sep_ptr = obj_sep_cstring.as_ptr();
717                let array_sep_ptr = array_sep_cstring.as_ptr();
718                (Some(obj_sep_cstring), Some(array_sep_cstring), obj_sep_ptr, array_sep_ptr)
719            } else {
720                (None, None, std::ptr::null(), std::ptr::null())
721            };
722
723        cpp!(unsafe [
724            schema_ptr as "const char*",
725            any_whitespace as "bool",
726            has_indent as "bool",
727            indent_value as "int",
728            has_separators as "bool",
729            obj_sep_ptr as "const char*",
730            array_sep_ptr as "const char*",
731            strict_mode as "bool",
732            has_max_whitespace_cnt as "bool",
733            max_whitespace_cnt_value as "int",
734            print_converted_ebnf as "bool"
735        ] -> Grammar as "xgrammar::Grammar" {
736            std::string schema_str(schema_ptr);
737            std::optional<int> opt_indent = has_indent ? std::make_optional(indent_value) : std::nullopt;
738            std::optional<std::pair<std::string, std::string>> opt_separators;
739
740            if (has_separators) {
741                opt_separators = std::make_pair(std::string(obj_sep_ptr), std::string(array_sep_ptr));
742            } else {
743                opt_separators = std::nullopt;
744            }
745
746            std::optional<int> opt_max_whitespace_cnt = has_max_whitespace_cnt ? std::make_optional(max_whitespace_cnt_value) : std::nullopt;
747
748            return xgrammar::Grammar::FromJSONSchema(
749                schema_str,
750                any_whitespace,
751                opt_indent,
752                opt_separators,
753                strict_mode,
754                opt_max_whitespace_cnt,
755                print_converted_ebnf
756            );
757        })
758    }

796    /// Construct a grammar from a structural tag JSON string.
797    ///
798    /// This method creates a grammar from structural tags that enable grammar-guided generation
799    /// with specific formatting markers. Structural tags are useful for dispatching between
800    /// different grammars based on trigger tokens and wrapping content with specific begin/end tags.
801    ///
802    /// The structural tag format supports:
803    /// - Single tag specification with begin marker, JSON schema, and end marker
804    /// - Multiple tags for grammar dispatching based on triggers
805    /// - Legacy tag/trigger pattern support
806    ///
807    /// # Arguments
808    /// * `structural_tag_json` - A JSON string specifying the structural tag configuration.
809    ///   The JSON should contain structural tag items with `begin`, `schema`, and `end` fields,
810    ///   and optionally `triggers` for grammar dispatching.
811    ///
812    /// # Returns
813    /// * `Ok(Grammar)` if the JSON is valid and the grammar was successfully created
814    /// * `Err(XGrammarErr)` if the JSON is invalid or the structural tag is malformed
815    ///
816    /// # Example
817    /// ```
818    /// # use xgrammar::Grammar;
819    /// use serde_json::json;
820    ///
821    /// // Triggered tags example for tool calling with multiple functions
822    /// let structural_tag = json!({
823    ///     "format": {
824    ///         "type": "triggered_tags",
825    ///         "triggers": ["<function="],
826    ///         "tags": [
827    ///             {
828    ///                 "begin": "<function=get_weather>",
829    ///                 "content": {
830    ///                     "type": "json_schema",
831    ///                     "json_schema": {
832    ///                         "type": "object",
833    ///                         "properties": {
834    ///                             "city": {"type": "string"},
835    ///                             "unit": {"type": "string", "enum": ["celsius", "fahrenheit"]}
836    ///                         },
837    ///                         "required": ["city"]
838    ///                     }
839    ///                 },
840    ///                 "end": "</function>"
841    ///             }
842    ///         ]
843    ///     }
844    /// });
845    ///
846    /// let grammar = Grammar::from_structural_tag(&structural_tag.to_string()).unwrap();
847    /// assert!(!grammar.is_null());
848    /// ```
849    pub fn from_structural_tag(structural_tag_json: &str) -> Result<Self> {
850        let structural_tag_json_cstring = CString::new(structural_tag_json)
851            .expect("Failed to convert structural_tag_json to CString");
852        let structural_tag_json_ptr = structural_tag_json_cstring.as_ptr();
853
854        // We use separate variables to capture success flag and error message
855        let mut success: i32 = 0;
856        let mut error_msg_ptr: *mut std::os::raw::c_char = std::ptr::null_mut();
857
858        let grammar = cpp!(unsafe [
859            structural_tag_json_ptr as "const char*",
860            mut success as "int",
861            mut error_msg_ptr as "char*"
862        ] -> Grammar as "xgrammar::Grammar" {
863            std::string structural_tag_json_str(structural_tag_json_ptr);
864            auto result = xgrammar::Grammar::FromStructuralTag(structural_tag_json_str);
865
866            // Check if result holds a Grammar or an error
867            if (std::holds_alternative<xgrammar::Grammar>(result)) {
868                success = 1;
869                error_msg_ptr = nullptr;
870                return std::get<xgrammar::Grammar>(result);
871            } else {
872                success = 0;
873                auto error = std::get<xgrammar::StructuralTagError>(result);
874
875                // Extract error message from the variant
876                std::string error_msg;
877                std::visit([&error_msg](auto&& err) {
878                    error_msg = err.what();
879                }, error);
880
881                // Allocate and copy error message
882                error_msg_ptr = strdup(error_msg.c_str());
883
884                // Return a null grammar
885                return xgrammar::Grammar(xgrammar::NullObj{});
886            }
887        });
888
889        if success == 1 {
890            Ok(grammar)
891        } else {
892            let error_msg = if !error_msg_ptr.is_null() {
893                let c_str = unsafe { CString::from_raw(error_msg_ptr) };
894                c_str.to_string_lossy().into_owned()
895            } else {
896                "Unknown error".to_string()
897            };
898            Err(XGrammarErr::InvalidStructuralTag(error_msg))
899        }
900    }

923    /// Create a grammar that matches any of the provided grammars.
924    ///
925    /// This method combines multiple grammars using a union operation, creating a new grammar
926    /// that accepts input matching any of the input grammars. This is equivalent to the `|`
927    /// (OR) operator in regular expressions.
928    ///
929    /// # Arguments
930    /// * `grammars` - A slice of Grammar objects to combine
931    ///
932    /// # Returns
933    /// * A new Grammar that matches if any of the input grammars match
934    ///
935    /// # Example
936    /// ```
937    /// # use xgrammar::Grammar;
938    /// let grammar1 = Grammar::from_regex(r"[0-9]+", Some(false));
939    /// let grammar2 = Grammar::from_regex(r"[a-z]+", Some(false));
940    /// let union_grammar = Grammar::union(&[grammar1, grammar2]);
941    /// assert!(!union_grammar.is_null());
942    /// ```
943    pub fn union(grammars: &[Grammar]) -> Self {
944        let grammars_ptr = grammars.as_ptr();
945        let num_grammars = grammars.len();
946        cpp!(unsafe [
947            grammars_ptr as "const xgrammar::Grammar*",
948            num_grammars as "size_t"
949        ] -> Grammar as "xgrammar::Grammar" {
950            std::vector<xgrammar::Grammar> grammars_vec;
951            grammars_vec.reserve(num_grammars);
952            for (size_t i = 0; i < num_grammars; ++i) {
953                grammars_vec.push_back(grammars_ptr[i]);
954            }
955            return xgrammar::Grammar::Union(grammars_vec);
956        })
957    }

1026    /// Create a GrammarMatcher from a compiled grammar.
1027    /// # Arguments
1028    /// * `compiled_grammar` - The compiled grammar to use
1029    /// * `override_stop_tokens` - Optional list of token ids to override the default stop tokens
1030    /// * `terminate_without_stop_token` - Whether to terminate the matcher without accepting a stop token.
1031    /// * `max_rollback_tokens` - Deprecated. You don't need to set it and it's always unlimited (-1).
1032    ///   The new Earley parser significantly reduces the number of states, so we can allow
1033    ///   unlimited rollback. The maximum number of rollback tokens allowed. The rollback operation
1034    ///   is useful for jump-forward decoding and speculative decoding.
1035    pub fn with(
1036        compiled_grammar: &CompiledGrammar,
1037        override_stop_tokens: Option<&[i32]>,
1038        terminate_without_stop_token: Option<bool>,
1039        max_rollback_tokens: Option<i32>,
1040    ) -> Self {
1041        // Keep it sync with the C++ implementation:
1042        // https://github.com/mlc-ai/xgrammar/blob/95bdfce011506ea95306b37d080115a2da3e369a/cpp/grammar_matcher.cc#L257
1043        let terminate_without_stop_token = terminate_without_stop_token.unwrap_or(false);
1044        let max_rollback_tokens = max_rollback_tokens.unwrap_or(0);
1045        let override_stop_tokens_ptr =
1046            override_stop_tokens.as_ref().map_or(std::ptr::null(), |v| v.as_ptr());
1047        let override_stop_tokens_len = override_stop_tokens.as_ref().map_or(0, |v| v.len());
1048
1049        cpp!(unsafe [
1050            compiled_grammar as "const xgrammar::CompiledGrammar*",
1051            override_stop_tokens_ptr as "const int32_t*",
1052            override_stop_tokens_len as "size_t",
1053            terminate_without_stop_token as "bool",
1054            max_rollback_tokens as "int"
1055        ] -> GrammarMatcher as "xgrammar::GrammarMatcher" {
1056            std::optional<std::vector<int32_t>> opt_override_stop_tokens;
1057            if (override_stop_tokens_len > 0) {
1058                opt_override_stop_tokens = std::vector<int32_t>(
1059                    *override_stop_tokens_ptr,
1060                    *override_stop_tokens_ptr + override_stop_tokens_len
1061                );
1062            } else {
1063                opt_override_stop_tokens = std::nullopt;
1064            }
1065
1066            return xgrammar::GrammarMatcher(
1067                *compiled_grammar,
1068                opt_override_stop_tokens,
1069                terminate_without_stop_token,
1070                max_rollback_tokens
1071            );
1072        })
1073    }

1119    /// Get the set of tokens that are acceptable for the next step and store them in a bitmask.
1120    ///
1121    /// # Arguments
1122    /// * `next_token_bitmask` - The bitmask to store the result. The bitmask must be pre-allocated
1123    ///   a DLTensor with shape (tokenizer.GetVocabSize() + 31) / 32, and dtype int32.
1124    /// * `index` - The index of the bitmask to fill. If None, the first bitmask is filled.
1125    /// * `debug_print` - If true, print debug information.
1126    ///
1127    /// # Returns
1128    /// * Whether the bitmask need to be applied (not all-true).
1129    pub fn fill_next_token_bitmask(
1130        &mut self,
1131        next_token_bitmask: &mut DLTensor,
1132        index: Option<usize>,
1133        debug_print: Option<bool>,
1134    ) -> bool {
1135        let dl_tensor = next_token_bitmask.dl_tensor();
1136        let index = index.unwrap_or(0) as i32;
