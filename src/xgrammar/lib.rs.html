2//! Safe, idiomatic Rust bindings for the [xgrammar](https://github.com/mlc-ai/xgrammar)
3//! C++ library for constrained decoding of large language models.
4//!
5//! This crate wraps xgrammar's grammar compilation and token-level matching so
6//! you can drive constrained generation (JSON schema, regex, BNF, structural
7//! tags) from Rust while retaining the performance of the upstream C++
8//! implementation.
9//!
10//! # Highlights
11//!
12//! - [`Grammar`], [`GrammarCompiler`], [`CompiledGrammar`], [`TokenizerInfo`] —
13//!   compile grammars (BNF, JSON schema, regex, structural tags) against a
14//!   tokenizer.
15//! - [`GrammarMatcher`] — token-by-token constrained decoding, including
16//!   [`GrammarMatcher::is_completed`] (root-rule match without stop token),
17//!   [`GrammarMatcher::fork`] for speculative / branching decoding, and
18//!   [`GrammarMatcher::traverse_draft_tree`] for filling per-node token
19//!   bitmasks over a speculative-decoding draft tree.
20//! - [`BatchGrammarMatcher`] — batched helpers over a slice of matchers:
21//!   [`BatchGrammarMatcher::batch_fill_next_token_bitmask`] is parallel and
22//!   thread-pool-backed; `batch_accept_token` / `batch_accept_string` /
23//!   `batch_rollback` are sequential static helpers.
24//! - Serialization — `serialize_json` / `deserialize_json` on [`Grammar`],
25//!   [`CompiledGrammar`], and [`TokenizerInfo`] for persisting compilation
26//!   results (version-locked to the vendored xgrammar's serialization format).
27//! - Typed errors — [`XGrammarErr`] variants mirror the upstream xgrammar
28//!   exception types via `XGrammarError::GetType()`
29//!   ([`XGrammarErr::InvalidJson`], [`XGrammarErr::InvalidStructuralTag`],
30//!   [`XGrammarErr::DeserializeVersion`], [`XGrammarErr::DeserializeFormat`],
31//!   ...), with the coarse-grained variants kept as fallbacks for untyped C++
32//!   errors.
33//!
34//! See each item's documentation for usage details, including when a
35//! `BatchGrammarMatcher` instance is required vs when associated functions can
36//! be called directly.

74    // Discriminants for the xgrammar::XGrammarError subclasses. Keep the
75    // numeric values in sync with the match in the Rust `error_from_kind`
76    // function (defined alongside the result-struct mirrors).
77    enum XGErrorKind : int32_t {
78        kXGOtherError = 0,
79        kXGDeserializeVersionError = 1,
80        kXGInvalidJSONError = 2,
81        kXGDeserializeFormatError = 3,
82        kXGInvalidJSONSchemaError = 4,
83        kXGInvalidStructuralTagError = 5,
84    };
85
86    // Classify any exception into an XGErrorKind. dynamic_cast (rather than
87    // comparing XGrammarError::GetType() strings) so that an upstream class
88    // rename breaks compilation loudly instead of silently degrading every
89    // typed error to kXGOtherError. Taking std::exception lets a single
90    // catch clause handle both typed and untyped errors.
91    static int32_t xgr_error_kind(const std::exception& e) {
92        if (dynamic_cast<const xgrammar::DeserializeVersionError*>(&e)) {
93            return kXGDeserializeVersionError;
94        }
95        if (dynamic_cast<const xgrammar::InvalidJSONError*>(&e)) {
96            return kXGInvalidJSONError;
97        }
98        if (dynamic_cast<const xgrammar::DeserializeFormatError*>(&e)) {
99            return kXGDeserializeFormatError;
100        }
101        if (dynamic_cast<const xgrammar::InvalidJSONSchemaError*>(&e)) {
102            return kXGInvalidJSONSchemaError;
103        }
104        if (dynamic_cast<const xgrammar::InvalidStructuralTagError*>(&e)) {
105            return kXGInvalidStructuralTagError;
106        }
107        return kXGOtherError;
108    }
109
110    // Extract the message and typed error kind from a std::variant of
111    // XGrammarError subclasses (xgrammar::SerializationError or
112    // xgrammar::StructuralTagError). The visitor can take the base reference
113    // because every variant alternative derives from XGrammarError.
114    template <typename V>
115    static char* xgr_dup_variant_error(const V& error, int32_t* error_kind) {
116        std::string error_msg;
117        std::visit([&](const xgrammar::XGrammarError& err) {
118            error_msg = err.what();
119            *error_kind = xgr_error_kind(err);
120        }, error);
121        return strdup(error_msg.c_str());
122    }

125    // mirror on the Rust side. `error_kind` (an XGErrorKind value) is

192/// Helper function to safely extract and free C++ error message.
193///
194/// # Safety
195/// The error_message_ptr must be null or a valid C string pointer allocated
196/// with strdup.
197unsafe fn extract_and_free_error_message(error_message_ptr: *mut std::os::raw::c_char) -> String {
198    // Every failure path fills the pointer with strdup(), which returns NULL
199    // when allocation fails — degrade to a placeholder instead of passing
200    // NULL to CStr::from_ptr.
201    if error_message_ptr.is_null() {
202        return "<error message unavailable: allocation failure>".to_string();
203    }
204    // SAFETY: The caller guarantees that error_message_ptr is a valid C string
205    // allocated with strdup; the null case is handled above.
206    unsafe {
207        let msg = CStr::from_ptr(error_message_ptr).to_string_lossy().into_owned();
208        libc::free(error_message_ptr as *mut libc::c_void);
209        msg
210    }
211}

213/// Copy a C++-owned byte buffer into `*out` as a Rust `String`. Shared body of
214/// the `rust!` callbacks in the `serialize_json` bindings.
215///
216/// The buffer is marshaled as `*const u8` (not `c_char`) so no platform-varying
217/// cast is needed (`c_char` is `i8` on x86_64 but `u8` on arm64). The upstream
218/// serializer maps raw bytes to Latin-1 code points (`ByteToLatin1`), so the
219/// JSON text is valid UTF-8 by construction; the lossy conversion is defensive
220/// only.
221///
222/// # Safety
223/// `data`/`len` must describe a readable buffer that outlives the call, and
224/// `out` must point to a live `String`.
225unsafe fn assign_utf8_lossy(out: *mut String, data: *const u8, len: usize) {
226    let slice = unsafe { std::slice::from_raw_parts(data, len) };
227    unsafe { *out = String::from_utf8_lossy(slice).into_owned() };
228}

230/// Ensure a DLPack tensor is row-major contiguous before handing it to
231/// xgrammar, which indexes `data` as a compact row-major array and ignores
232/// `DLTensor::strides` entirely. A permuted or sliced layout would be read
233/// and written at wrong offsets (silently corrupting results), and a
234/// negative stride would place accesses outside the buffer.
235fn ensure_row_major_contiguous(tensor: &DLTensor, name: &str) -> Result<()> {
236    if tensor.memory_order() == dlpark::utils::MemoryOrder::RowMajorContiguous {
237        return Ok(());
238    }
239    Err(XGrammarErr::MatcherError(format!(
240        "The {} tensor must be row-major contiguous: xgrammar ignores DLTensor strides \
241         (shape {:?}, strides {:?})",
242        name,
243        tensor.shape(),
244        tensor.strides(),
245    )))
246}

248/// View a DLPack tensor as an `i64` slice when it is a contiguous 1-D int64
249/// CPU tensor; `None` otherwise (the C++ side then reports its own
250/// validation error for the malformed tensor). The device check is
251/// load-bearing: a non-CPU data pointer must not be dereferenced on the
252/// host.
253fn as_cpu_i64_slice(tensor: &DLTensor) -> Option<&[i64]> {
254    let dtype = tensor.data_type();
255    if tensor.dl_tensor().device.device_type != dlpark::ffi::DeviceType::Cpu
256        || tensor.num_dimensions() != 1
257        || dtype.code != dlpark::ffi::DataTypeCode::Int
258        || dtype.bits != 64
259        || dtype.lanes != 1
260    {
261        return None;
262    }
263    tensor.as_slice_contiguous::<i64>().ok()
264}

266/// Validate a draft-tree encoding before handing the tensors to xgrammar:
267/// upstream `TraverseDraftTree` follows `retrieve_next_token` /
268/// `retrieve_next_sibling` indices without any bounds or cycle check, so an
269/// out-of-range link would corrupt memory and a cycle would recurse forever.
270/// Every link must stay in `[-1, N)` and each node may be reached at most
271/// once (the encoding must be a tree).
272fn validate_draft_tree(next_token: &[i64], next_sibling: &[i64]) -> Result<()> {
273    let n = next_token.len();
274    if n == 0 {
275        return Ok(()); // The C++ side rejects empty tensors itself.
276    }
277    let mut visited = vec![false; n];
278    let mut stack = vec![0usize];
279    while let Some(node) = stack.pop() {
280        if visited[node] {
281            return Err(XGrammarErr::MatcherError(format!(
282                "Invalid draft tree: node {node} is reachable more than once \
283                 (the encoding must be an acyclic tree)"
284            )));
285        }
286        visited[node] = true;
287        for link in [next_token[node], next_sibling[node]] {
288            if link == -1 {
289                continue;
290            }
291            if link < 0 || link >= n as i64 {
292                return Err(XGrammarErr::MatcherError(format!(
293                    "Invalid draft tree: node {node} links to index {link}, outside [-1, {n})"
294                )));
295            }
296            stack.push(link as usize);
297        }
298    }
299    Ok(())
300}

302/// Map a C++ `XGErrorKind` discriminant + message to a precise
303/// [`XGrammarErr`], falling back to the per-result-type context variant when
304/// the exception was untyped (kind 0, e.g. a plain `std::runtime_error` from
305/// `XGRAMMAR_CHECK`). Keep the discriminants in sync with `XGErrorKind` in the
306/// `cpp!{{...}}` block above; an out-of-range value degrades to the fallback.
307fn error_from_kind(raw_kind: i32, msg: String, fallback: fn(String) -> XGrammarErr) -> XGrammarErr {
308    match raw_kind {
309        1 => XGrammarErr::DeserializeVersion(msg),
310        2 => XGrammarErr::InvalidJson(msg),
311        3 => XGrammarErr::DeserializeFormat(msg),
312        4 => XGrammarErr::InvalidJsonSchema(msg),
313        5 => XGrammarErr::InvalidStructuralTag(msg),
314        _ => fallback(msg),
315    }
316}

493impl TokenizerInfo {
494    pub fn from_backend_str(
495        backend_str: &str,
496        vocab_size: Option<usize>,
497        stop_token_ids: Vec<TokenId>,
498    ) -> self::Result<Self> {
499        let tokenizer = tokenizers::Tokenizer::from_str(backend_str).map_err(|e| {
500            XGrammarErr::TokenizerParseFailed(format!("failed to parse tokenizer: {}", e))
501        })?;
502        let vocab_map = tokenizer.get_vocab(true); // with added special tokens
503        let max_id = vocab_map
504            .values()
505            .max()
506            .ok_or(XGrammarErr::InvalidTokenizerConfig("Vocab map is empty".to_string()))?;
507        let tokenizer_vocab_size = std::cmp::max(vocab_map.len(), (max_id + 1) as usize);
508        if let Some(vocab_size) = vocab_size
509            && vocab_size != tokenizer_vocab_size
510        {
511            tracing::warn!(
512                "Provided vocab_size {} does not match tokenizer vocab size {}. Using provided vocab_size.",
513                vocab_size,
514                tokenizer_vocab_size
515            );
516        }
517        let final_vocab_size = vocab_size.unwrap_or(tokenizer_vocab_size);
518        let tokenizer_metadata = Self::detect_metadata_from_hf(backend_str)?;
519        let vocab_type = tokenizer_metadata.vocab_type;
520        let add_prefix_space = tokenizer_metadata.add_prefix_space;
521
522        Self::new(vocab_map, vocab_type, final_vocab_size, stop_token_ids, add_prefix_space)
523    }

525    pub fn parse_eos_token(path: &Path, json_key: &str) -> Option<Vec<i32>> {
526        let contents = std::fs::read_to_string(path).ok()?;
527        let json: Value = serde_json::from_str(&contents).ok()?;
528        match json.get(json_key) {
529            Some(Value::Number(num)) if num.is_i64() => Some(vec![num.as_i64().unwrap() as i32]),
530            Some(Value::Array(arr)) => {
531                let mut eos_tokens = Vec::new();
532                for item in arr {
533                    if let Value::Number(num) = item {
534                        if num.is_i64() {
535                            eos_tokens.push(num.as_i64().unwrap() as i32);
536                        } else {
537                            return None;
538                        }
539                    } else {
540                        return None;
541                    }
542                }
543                Some(eos_tokens)
544            }
545            _ => None,
546        }
547    }

549    pub fn from_path<P>(
550        path: P,
551        vocab_size: Option<usize>,
552        stop_token_ids: Option<Vec<TokenId>>,
553    ) -> Result<Self>
554    where
555        P: AsRef<Path>,
556    {
557        let path = path.as_ref();
558        let tokenizer_json_path = path.join(TOKENIZER_FILE);
559        let backend_str = std::fs::read_to_string(&tokenizer_json_path)
560            .map_err(XGrammarErr::TokenizerLoadFailed)?;
561
562        let eos_token = Self::parse_eos_token(&path.join(GENERATION_CONFIG_FILE), EOS_TOKEN_ID_KEY)
563            .or_else(|| Self::parse_eos_token(&path.join(HF_CONFIG_FILE), EOS_TOKEN_ID_KEY));
564
565        let mut stop_token_ids = stop_token_ids.unwrap_or_default();
566        stop_token_ids.extend(eos_token.unwrap_or_default());
567        stop_token_ids.dedup();
568
569        Self::from_backend_str(&backend_str, vocab_size, stop_token_ids)
570    }

596    fn new(
597        vocab_map: HashMap<String, u32>,

602    ) -> self::Result<Self> {
603        // Ensure the vocab size is at least as large as the max id in the vocab map.
604        // Marshal each token as pointer + length (not CString): tokens are
605        // arbitrary byte sequences and may legally contain NUL bytes, which
606        // upstream's std::vector<std::string> preserves.
607        let mut encoded_vocab: Vec<&str> = vec![""; vocab_size];
608
609        // Fill the encoded_vocab with tokens from the vocab_map
610        for (token, idx) in vocab_map.iter() {
611            assert!(
612                (*idx as usize) < vocab_size,
613                "Token ID {} exceeds vocab size {}",
614                idx,
615                vocab_size
616            );
617            encoded_vocab[*idx as usize] = token.as_str();
618        }
619
620        let token_ptrs: Vec<*const u8> = encoded_vocab.iter().map(|s| s.as_ptr()).collect();
621        let token_lens: Vec<usize> = encoded_vocab.iter().map(|s| s.len()).collect();
622        let token_ptrs_ptr = token_ptrs.as_ptr();
623        let token_lens_ptr = token_lens.as_ptr();
624        let vocab_size_i32 = vocab_size as i32;
625        let stop_token_ids_ptr = stop_token_ids.as_ptr();
626        let stop_token_ids_len = stop_token_ids.len();
627
628        Ok(cpp!(unsafe [
629            token_ptrs_ptr as "const uint8_t* const*",
630            token_lens_ptr as "const size_t*",
631            vocab_type as "xgrammar::VocabType",
632            vocab_size_i32 as "int",
633            stop_token_ids_ptr as "const int32_t*",
634            stop_token_ids_len as "size_t",
635            add_prefix_space as "bool"
636        ] -> TokenizerInfo as "xgrammar::TokenizerInfo" {
637            std::vector<std::string> encoded_vocab;
638            encoded_vocab.reserve(vocab_size_i32);
639            for (int i = 0; i < vocab_size_i32; ++i) {
640                encoded_vocab.emplace_back(
641                    reinterpret_cast<const char*>(token_ptrs_ptr[i]), token_lens_ptr[i]
642                );
643            }
644            std::vector<int32_t> stop_token_ids(stop_token_ids_ptr, stop_token_ids_ptr + stop_token_ids_len);
645
646            return xgrammar::TokenizerInfo(
647                encoded_vocab,
648                vocab_type,
649                vocab_size_i32,
650                stop_token_ids,
651                add_prefix_space
652            );
653        }))
654    }

677    // const std::vector<std::string>& GetDecodedVocab() const;
678    pub fn get_decoded_vocab(&self) -> Vec<String> {
679        // Avoid relying on layout-compatibility between `Vec<T>` and `std::vector<T>`
680        // (Rust Vec is (ptr, cap, len); libstdc++ std::vector is (start, finish,
681        // end_of_storage) — different semantics for the second/third word). Instead
682        // the C++ side writes each element into a Rust-allocated `Vec<String>` via
683        // the `vec_push_string` callback bridge.
684        let mut out: Vec<String> = Vec::new();
685        let out_ptr = &mut out as *mut Vec<String>;
686        cpp!(unsafe [
687            self as "const xgrammar::TokenizerInfo*",
688            out_ptr as "void*"
689        ] {
690            const auto& vocab = self->GetDecodedVocab();
691            for (const auto& s : vocab) {
692                // Marshal as `uint8_t*` so the Rust side receives `*const u8`
693                // directly, avoiding a `c_char`→`u8` cast whose necessity
694                // varies by platform (c_char is i8 on x86_64 but u8 on arm64,
695                // which makes the cast trigger `clippy::unnecessary_cast`
696                // on arm64).
697                const uint8_t* data = reinterpret_cast<const uint8_t*>(s.data());
698                size_t len = s.size();
699                rust!(XGR_TokInfo_DecodedVocab_push [
700                    out_ptr: *mut Vec<String> as "void*",
701                    data: *const u8 as "const uint8_t*",
702                    len: usize as "size_t"
703                ] {
704                    // SAFETY: `data`/`len` point into the C++ std::string; the
705                    // slice is only read during this call. `out_ptr` was
706                    // obtained from a live `&mut Vec<String>` on the Rust side.
707                    let slice = unsafe { std::slice::from_raw_parts(data, len) };
708                    // Must be `from_utf8_lossy`, not `from_utf8_unchecked`:
709                    // xgrammar's ByteFallback / ByteLevel decoders (see
710                    // thirdparty/xgrammar/cpp/tokenizer_info.cc `DecodeToken`)
711                    // can return single raw bytes (e.g. `<0x80>` → 0x80), which
712                    // are not valid UTF-8. `unchecked` would be UB here.
713                    let s = String::from_utf8_lossy(slice).into_owned();
714                    unsafe { (*out_ptr).push(s) };
715                });
716            }
717        });
718        out
719    }

795        let backend_str_len = backend_str.len();
796        // Error out-parameter: set to a strdup'd message when the C++ side
797        // fails. C++ exceptions must not cross the FFI boundary, so the
798        // closure catches everything and reports through this channel.
799        let mut error_message: *mut std::os::raw::c_char = std::ptr::null_mut();
800        let error_message_out = &mut error_message as *mut *mut std::os::raw::c_char;
801
802        let metadata = cpp!(unsafe [
803            backend_str_ptr as "const uint8_t*",
804            backend_str_len as "size_t",
805            error_message_out as "char**"
806        ] -> MetadataFromHF as "MetadataFromHF" {
807            try {
808                std::string backend_str(
809                    reinterpret_cast<const char*>(backend_str_ptr), backend_str_len
810                );
811                std::string metadata_str = TokenizerInfo::DetectMetadataFromHF(backend_str);
812                picojson::value v;
813                std::string err = picojson::parse(v, metadata_str);
814                if (!err.empty()) {
815                    throw std::runtime_error("Failed to parse metadata: " + err);
816                }
817                const picojson::object& metadata = v.get<picojson::object>();
818
819                MetadataFromHF metadata_from_hf;
820                metadata_from_hf.vocab_type = static_cast<xgrammar::VocabType>(metadata["vocab_type"].get<double>());
821                metadata_from_hf.add_prefix_space = metadata["add_prefix_space"].get<bool>();
822                return metadata_from_hf;
823            } catch (const std::exception& e) {
824                *error_message_out = strdup(e.what());
825                return MetadataFromHF{VocabType::RAW, false};
826            }
827        });
828
829        if !error_message.is_null() {
830            // SAFETY: non-null means the C++ side just strdup'd it.
831            let msg = unsafe { extract_and_free_error_message(error_message) };
832            return Err(XGrammarErr::TokenizerParseFailed(msg));
833        }
834        Ok(metadata)
835    }

924

2063    /// Traverse a draft token tree (speculative decoding) and fill one token
2064    /// bitmask row per tree node.
2065    ///
2066    /// Performs a depth-first traversal of the draft tree: row `i` of
2067    /// `token_bitmask` receives the set of tokens allowed *after* accepting the
2068    /// draft tokens on the path from the root to node `i`. Verifying an entire
2069    /// draft tree this way is much cheaper than replaying
2070    /// [`Self::accept_token`] / [`Self::fill_next_token_bitmask`] /
2071    /// [`Self::rollback`] per node from Rust.
2072    ///
2073    /// # Tree encoding
2074    /// Node `0` is the root. `retrieve_next_token[i]` is the index of node
2075    /// `i`'s first child, or `-1` if it has none; `retrieve_next_sibling[i]` is
2076    /// the index of node `i`'s next sibling, or `-1`. The root must not have a
2077    /// sibling (`retrieve_next_sibling[0] == -1`).
2078    ///
2079    /// `draft_tokens[0]` is ignored: the root token is the one already produced
2080    /// by the target model, so its acceptance is not re-checked and row `0` is
2081    /// simply the bitmask of the matcher's current state.
2082    ///
2083    /// # Arguments
2084    /// * `retrieve_next_token` - 1-D `int64` CPU tensor of first-child indices.
2085    /// * `retrieve_next_sibling` - 1-D `int64` CPU tensor of sibling indices.
2086    /// * `draft_tokens` - 1-D `int64` CPU tensor of the draft token id at each node.
2087    ///   All three tensors must have the same length `N >= 1` (the node count).
2088    /// * `token_bitmask` - Pre-allocated 2-D `int32` CPU tensor of shape
2089    ///   `(N, get_bitmask_size(vocab_size))` (see [`get_bitmask_size`]).
2090    /// * `time_threshold` - Maximum traversal time in seconds; `None` (or a
2091    ///   value `<= 0`) disables the timeout.
2092    ///
2093    /// # Node semantics
2094    /// A draft token that is out of vocabulary range or not permitted by its
2095    /// parent's bitmask row gets its row zeroed, and its **entire subtree is
2096    /// skipped without touching the descendants' rows** (they keep whatever the
2097    /// caller pre-filled). A node whose acceptance *terminates* the matcher is
2098    /// treated the same way: its row is zeroed (no token may follow) and its
2099    /// subtree is skipped. In both cases traversal continues with the node's
2100    /// siblings and still returns `Ok(true)`. Calling this method on an
2101    /// already-terminated matcher likewise returns `Ok(true)` with row `0`
2102    /// zeroed and every other row untouched (unlike
2103    /// [`Self::fill_next_token_bitmask`], which returns an error).
2104    ///
2105    /// The C++ traversal recurses once per tree level, so extremely deep
2106    /// chains (hundreds of thousands of nodes in a line) can exhaust the
2107    /// native stack.
2108    ///
2109    /// # Returns
2110    /// * `Ok(true)` - The traversal completed. The matcher state is unchanged
2111    ///   (internal accepts and rollbacks cancel out).
2112    /// * `Ok(false)` - The traversal timed out. Row `0` is always filled (the
2113    ///   timeout is only checked at non-root nodes) and unvisited rows keep
2114    ///   whatever the caller pre-filled, but the matcher state is fully
2115    ///   restored — every internal accept is rolled back as the failure
2116    ///   propagates — so the matcher can keep being used without a
2117    ///   [`Self::reset`].
2118    /// * `Err(XGrammarErr::MatcherError)` - A tensor has an invalid dtype,
2119    ///   shape, or device type, is not row-major contiguous, or the tree
2120    ///   encoding is invalid (a link outside `[-1, N)`, or a node reachable
2121    ///   more than once).
2122    ///
2123    /// # Example
2124    /// ```no_run
2125    /// # use xgrammar::{get_bitmask_size, CompiledGrammar, GrammarMatcher};
2126    /// # use dlpark::versioned::SafeManagedTensorVersioned;
2127    /// # use ndarray::{ArrayD, IxDyn};
2128    /// # fn example(compiled: &CompiledGrammar) -> xgrammar::Result<()> {
2129    /// let mut matcher = GrammarMatcher::new(compiled);
2130    /// let vocab_size = compiled.get_tokenizer_info().get_vocab_size();
2131    /// let bitmask_len = get_bitmask_size(vocab_size) as usize;
2132    ///
2133    /// // Linear draft tree: node 0 (root) -> node 1 -> node 2.
2134    /// let next_token = SafeManagedTensorVersioned::new(vec![1i64, 2, -1]).unwrap();
2135    /// let next_sibling = SafeManagedTensorVersioned::new(vec![-1i64, -1, -1]).unwrap();
2136    /// let draft_tokens = SafeManagedTensorVersioned::new(vec![0i64, 123, 456]).unwrap();
2137    /// let bitmask = ArrayD::from_elem(IxDyn(&[3, bitmask_len]), 0i32);
2138    /// let mut bitmask = SafeManagedTensorVersioned::new(bitmask).unwrap();
2139    ///
2140    /// let completed = matcher.traverse_draft_tree(
2141    ///     &next_token, &next_sibling, &draft_tokens, &mut bitmask, None,
2142    /// )?;
2143    /// assert!(completed);
2144    /// # Ok(())
2145    /// # }
2146    /// ```
2147    pub fn traverse_draft_tree(
2148        &mut self,
2149        retrieve_next_token: &DLTensor,
2150        retrieve_next_sibling: &DLTensor,
2151        draft_tokens: &DLTensor,
2152        token_bitmask: &mut DLTensor,
2153        time_threshold: Option<f64>,
2154    ) -> Result<bool> {
2155        ensure_row_major_contiguous(retrieve_next_token, "retrieve_next_token")?;
2156        ensure_row_major_contiguous(retrieve_next_sibling, "retrieve_next_sibling")?;
2157        ensure_row_major_contiguous(draft_tokens, "draft_tokens")?;
2158        ensure_row_major_contiguous(token_bitmask, "token_bitmask")?;
2159        // The C++ traversal follows the tree links without any bounds or
2160        // cycle check, so validate them here. Tensors that are not 1-D int64
2161        // CPU are skipped: the C++ side rejects those itself before touching
2162        // any link.
2163        if let (Some(next_token), Some(next_sibling)) =

2301/// Batched helpers that operate on a slice of [`GrammarMatcher`].
2302///
2303/// ## When an instance is (and isn't) needed
2304///
2305/// The method layout intentionally mirrors upstream xgrammar, where only one batch
2306/// operation actually runs in parallel:
2307///
2308/// - [`Self::batch_fill_next_token_bitmask`] — **instance method** (`&mut self`).
2309///   Uses the thread pool owned by this `BatchGrammarMatcher` to fan the per-matcher
2310///   work out across threads. Construct once via [`Self::new`] or
2311///   [`Self::with_max_threads`] and reuse.
2312///
2313/// - [`Self::batch_accept_token`], [`Self::batch_accept_string`],
2314///   [`Self::batch_rollback`] — **associated (static) functions**.
2315///   Upstream implements these as a plain sequential `for` loop; no thread pool
2316///   or instance state is involved, so there is nothing for `self` to carry.
2317///   Call them as `BatchGrammarMatcher::batch_accept_token(...)` without
2318///   constructing an instance.
2319///
2320/// This asymmetry reflects the upstream implementation
2321/// (`BatchGrammarMatcher::Impl::BatchAcceptToken` etc. in
2322/// `thirdparty/xgrammar/cpp/grammar_matcher.cc`). If upstream ever parallelizes
2323/// those ops, they will gain a `&self` receiver here.
2324impl BatchGrammarMatcher {
2325    /// Create a `BatchGrammarMatcher` with the default `"auto"` thread policy
2326    /// (roughly half of the available hardware threads).
2327    ///
2328    /// The constructed instance owns a thread pool that is used **only** by
2329    /// [`Self::batch_fill_next_token_bitmask`]. If you do not call that method,
2330    /// you do not need a `BatchGrammarMatcher` instance — the other batch helpers
2331    /// are associated functions.
2332    ///
2333    /// Equivalent to `BatchGrammarMatcher::default()` (the `Default` impl is
2334    /// auto-generated by `cpp_class!` and maps to the C++ default constructor,
2335    /// which also uses `"auto"`).
2336    pub fn new() -> Self {
2337        cpp!(unsafe [] -> BatchGrammarMatcher as "xgrammar::BatchGrammarMatcher" {
2338            return xgrammar::BatchGrammarMatcher(std::string("auto"));
2339        })
2340    }

2342    /// Create a `BatchGrammarMatcher` with an explicit maximum thread count for
2343    /// the thread pool used by [`Self::batch_fill_next_token_bitmask`].

2357    /// Batched version of [`GrammarMatcher::fill_next_token_bitmask`].
2358    ///
2359    /// This is the **only** batch method that uses the thread pool of this
2360    /// `BatchGrammarMatcher`; it therefore takes `&mut self`. When
2361    /// `max_threads > 1` the per-matcher bitmask fills are executed in parallel.
2362    ///
2363    /// # Arguments
2364    /// * `matchers` - The matchers to operate on in parallel. Mutated in place.
2365    /// * `next_token_bitmask` - Pre-allocated `DLTensor` with shape `(N, bitmask_len)` and
2366    ///   dtype `int32`, where `N >= matchers.len()` and `bitmask_len` is the per-matcher
2367    ///   bitmask length.
2368    /// * `indices` - Optional mapping from matcher index to bitmask row. If `None`, the
2369    ///   bitmask row `i` is written for `matchers[i]`.
2370    /// * `debug_print` - When `true`, print debug information.
2371    pub fn batch_fill_next_token_bitmask(
2372        &mut self,
2373        matchers: &mut [GrammarMatcher],
2374        next_token_bitmask: &mut DLTensor,
2375        indices: Option<&[i32]>,
2376        debug_print: Option<bool>,
2377    ) -> Result<()> {
2378        ensure_row_major_contiguous(next_token_bitmask, "next_token_bitmask")?;
2379        let dl_tensor = next_token_bitmask.dl_tensor();
2380        let debug_print = debug_print.unwrap_or(false);
2381        let matchers_ptr = matchers.as_mut_ptr();
2382        let num_matchers = matchers.len();
2383        let indices_ptr = indices.map(|s| s.as_ptr()).unwrap_or(std::ptr::null());
2384        let num_indices = indices.map(|s| s.len()).unwrap_or(0);
2385        let has_indices = indices.is_some();
2386
2387        // xgrammar's batch API takes `std::vector<GrammarMatcher>*`, so we must
2388        // materialize a vector over our Rust-owned slice. This is cheap and safe
2389        // because `GrammarMatcher` is a shared_ptr<Impl> PIMPL (see upstream
2390        // xgrammar/object.h `XGRAMMAR_DEFINE_PIMPL_METHODS`): copying a matcher
2391        // clones the shared_ptr, so `matchers_vec[i]` aliases the *same* Impl as
2392        // `matchers_ptr[i]`. Batch ops mutate through `pimpl_`, so state changes
2393        // land in the shared Impl and are visible to the caller without any
2394        // write-back step.
2395        let result = cpp!(unsafe [
2396            self as "xgrammar::BatchGrammarMatcher*",
2397            matchers_ptr as "xgrammar::GrammarMatcher*",
2398            num_matchers as "size_t",
2399            dl_tensor as "DLTensor*",
2400            indices_ptr as "const int32_t*",
2401            num_indices as "size_t",
2402            has_indices as "bool",
2403            debug_print as "bool"
2404        ] -> MatcherResult as "MatcherResult" {
2405            try {
2406                std::vector<xgrammar::GrammarMatcher> matchers_vec(
2407                    matchers_ptr, matchers_ptr + num_matchers
2408                );
2409                std::optional<std::vector<int32_t>> opt_indices;
2410                if (has_indices) {
2411                    opt_indices = std::vector<int32_t>(
2412                        indices_ptr, indices_ptr + num_indices
2413                    );
2414                }
2415                self->BatchFillNextTokenBitmask(
2416                    &matchers_vec, dl_tensor, opt_indices, debug_print
2417                );
2418                return {true, false, nullptr, 0};
2419            } catch (const std::exception& e) {
2420                return {false, false, strdup(e.what()), xgr_error_kind(e)};
2421            }
2422        });

2424        result.into()
2425    }
