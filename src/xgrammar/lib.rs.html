97impl TokenizerInfo {
98    pub fn from_backend_str(
99        backend_str: &str,
100        vocab_size: Option<usize>,
101        stop_token_ids: Vec<TokenId>,
102    ) -> self::Result<Self> {
103        let tokenizer = tokenizers::Tokenizer::from_str(backend_str).map_err(|e| {
104            XGrammarErr::TokenizerParseFailed(format!("failed to parse tokenizer: {}", e))
105        })?;
106        let vocab_map = tokenizer.get_vocab(true); // with added special tokens
107        let max_id = vocab_map
108            .values()
109            .max()
110            .ok_or(XGrammarErr::InvalidTokenizerConfig("Vocab map is empty".to_string()))?;
111        let tokenizer_vocab_size = std::cmp::max(vocab_map.len(), (max_id + 1) as usize);
112        if let Some(vocab_size) = vocab_size {
113            if vocab_size != tokenizer_vocab_size {
114                tracing::warn!(
115                    "Provided vocab_size {} does not match tokenizer vocab size {}. Using provided vocab_size.",
116                    vocab_size,
117                    tokenizer_vocab_size
118                );
119            }
120        }
121        let final_vocab_size = vocab_size.unwrap_or(tokenizer_vocab_size);
122        let tokenizer_metadata = Self::detect_metadata_from_hf(backend_str);
123        let vocab_type = tokenizer_metadata.vocab_type;
124        let add_prefix_space = tokenizer_metadata.add_prefix_space;
125
126        Self::new(vocab_map, vocab_type, final_vocab_size, stop_token_ids, add_prefix_space)
127    }

129    pub fn parse_eos_token(path: &Path, json_key: &str) -> Option<Vec<i32>> {
130        let contents = std::fs::read_to_string(path).ok()?;
131        let json: Value = serde_json::from_str(&contents).ok()?;
132        match json.get(json_key) {
133            Some(Value::Number(num)) if num.is_i64() => Some(vec![num.as_i64().unwrap() as i32]),
134            Some(Value::Array(arr)) => {
135                let mut eos_tokens = Vec::new();
136                for item in arr {
137                    if let Value::Number(num) = item {
138                        if num.is_i64() {
139                            eos_tokens.push(num.as_i64().unwrap() as i32);
140                        } else {
141                            return None;
142                        }
143                    } else {
144                        return None;
145                    }
146                }
147                Some(eos_tokens)
148            }
149            _ => None,
150        }
151    }

153    pub fn from_path<P>(
154        path: P,
155        vocab_size: Option<usize>,
156        stop_token_ids: Option<Vec<TokenId>>,
157    ) -> Result<Self>
158    where
159        P: AsRef<Path>,
160    {
161        let path = path.as_ref();
162        let tokenizer_json_path = path.join(TOKENIZER_FILE);
163        let backend_str = std::fs::read_to_string(&tokenizer_json_path)
164            .map_err(XGrammarErr::TokenizerLoadFailed)?;
165
166        let eos_token = Self::parse_eos_token(&path.join(GENERATION_CONFIG_FILE), EOS_TOKEN_ID_KEY)
167            .or_else(|| Self::parse_eos_token(&path.join(HF_CONFIG_FILE), EOS_TOKEN_ID_KEY));
168
169        let mut stop_token_ids = stop_token_ids.unwrap_or_default();
170        stop_token_ids.extend(eos_token.unwrap_or_default());
171        stop_token_ids.dedup();
172
173        Self::from_backend_str(&backend_str, vocab_size, stop_token_ids)
174    }

176    #[cfg(feature = "hf_hub")]
177    pub fn from_pretrained(
178        tokenizer_id: &str,
179        revision: Option<String>,
180        vocab_size: Option<usize>,
181        stop_token_ids: Option<Vec<i32>>,
182    ) -> Result<TokenizerInfo> {
183        use huggingface_hub::{Params, Repo, RepoType, compile_glob_pattern, snapshot_download};
184
185        let allow_patterns = compile_glob_pattern(TOKENIZER_ALLOW_PATTERN).map_err(|e| {
186            XGrammarErr::TokenizerParseFailed(format!("Failed to compile glob patterns: {}", e))
187        })?;
188        let download_options =
189            Some(Params { allow_patterns: Some(allow_patterns), ..Default::default() });
190
191        let repo = Repo::with_revision(
192            tokenizer_id.to_string(),
193            RepoType::Model,
194            revision.unwrap_or("main".to_string()),
195        );
196        let tokenizer_dir = snapshot_download(repo, download_options)?;
197        Self::from_path(tokenizer_dir, vocab_size, stop_token_ids)
198    }

200    fn new(
201        vocab_map: HashMap<String, u32>,
202        vocab_type: VocabType,
203        vocab_size: usize,
204        stop_token_ids: Vec<i32>,
205        add_prefix_space: bool,
206    ) -> self::Result<Self> {
207        // Ensure the vocab size is at least as large as the max id in the vocab map
208        let mut encoded_vocab = vec![CString::new("").unwrap(); vocab_size];
209
210        // Fill the encoded_vocab with tokens from the vocab_map
211        for (token, idx) in vocab_map.iter() {
212            assert!(
213                (*idx as usize) < vocab_size,
214                "Token ID {} exceeds vocab size {}",
215                idx,
216                vocab_size
217            );
218            encoded_vocab[*idx as usize] =
219                CString::new(token.as_str()).expect("fail to convert a token to CString");
220        }
221
222        let encoded_vocab_ptr: Vec<_> = encoded_vocab.iter().map(|s| s.as_ptr()).collect();
223        let encoded_vocab_ptr_ptr = encoded_vocab_ptr.as_ptr();
224        let vocab_size_i32 = vocab_size as i32;
225        let stop_token_ids_ptr = stop_token_ids.as_ptr();
226        let stop_token_ids_len = stop_token_ids.len();
227
228        Ok(cpp!(unsafe [
229            encoded_vocab_ptr_ptr as "const char* const*",
230            vocab_type as "xgrammar::VocabType",
231            vocab_size_i32 as "int",
232            stop_token_ids_ptr as "const int32_t*",
233            stop_token_ids_len as "size_t",
234            add_prefix_space as "bool"
235        ] -> TokenizerInfo as "xgrammar::TokenizerInfo" {
236            std::vector<std::string> encoded_vocab;
237            for (int i = 0; i < vocab_size_i32; ++i) {
238                encoded_vocab.push_back(std::string(encoded_vocab_ptr_ptr[i]));
239            }
240            std::vector<int32_t> stop_token_ids(stop_token_ids_ptr, stop_token_ids_ptr + stop_token_ids_len);
241
242            return xgrammar::TokenizerInfo(
243                encoded_vocab,
244                vocab_type,
245                vocab_size_i32,
246                stop_token_ids,
247                add_prefix_space
248            );
249        }))
250    }

280    fn detect_metadata_from_hf(backend_str: &str) -> MetadataFromHF {
281        let backend_str =
282            CString::new(backend_str).expect("Failed to convert backend_str to CString");
283        let backend_str_ptr = backend_str.as_ptr();
284
285        cpp!(unsafe [backend_str_ptr as "const char*"] -> MetadataFromHF as "MetadataFromHF" {
286            const std::string &backend_str(backend_str_ptr);
287            std::string metadata_str = TokenizerInfo::DetectMetadataFromHF(backend_str);
288            picojson::value v;
289            std::string err = picojson::parse(v, metadata_str);
290            if (!err.empty()) {
291                throw std::runtime_error("Failed to parse metadata: " + err);
292            }
293            const picojson::object& metadata = v.get<picojson::object>();
294
295            MetadataFromHF metadata_from_hf;
296            metadata_from_hf.vocab_type = static_cast<xgrammar::VocabType>(metadata["vocab_type"].get<double>());
297            metadata_from_hf.add_prefix_space = metadata["add_prefix_space"].get<bool>();
298            return metadata_from_hf;
299        })
300    }

397    /// Get the compiled grammar for a JSON schema string.
398    ///
399    /// # Arguments
400    /// * `schema` - The JSON schema string to compile
401    /// * `any_whitespace` - Whether to allow any whitespace (default: true)
402    /// * `indent` - Optional indentation level
403    /// * `separators` - Optional custom separators (object_separator, array_separator)
404    /// * `strict_mode` - Whether to use strict mode (default: true)
405    ///
406    /// # Returns
407    /// * A compiled grammar that can be used with GrammarMatcher
408    pub fn compile_json_schema(
409        &self,
410        schema: &str,
411        any_whitespace: Option<bool>,
412        indent: Option<i32>,
413        separators: Option<(String, String)>,
414        strict_mode: Option<bool>,
415    ) -> CompiledGrammar {
416        let schema_cstring = CString::new(schema).expect("Failed to convert schema to CString");
417        let schema_ptr = schema_cstring.as_ptr();
418        let any_whitespace = any_whitespace.unwrap_or(true);
419        let strict_mode = strict_mode.unwrap_or(true);
420        let has_indent = indent.is_some();
421        let indent_value = indent.unwrap_or(0);
422        let has_separators = separators.is_some();
423
424        let (_obj_sep_cstring, _array_sep_cstring, obj_sep_ptr, array_sep_ptr) =
425            if let Some((obj_sep, array_sep)) = separators {
426                let obj_sep_cstring =
427                    CString::new(obj_sep).expect("Failed to convert object separator to CString");
428                let array_sep_cstring =
429                    CString::new(array_sep).expect("Failed to convert array separator to CString");
430                let obj_sep_ptr = obj_sep_cstring.as_ptr();
431                let array_sep_ptr = array_sep_cstring.as_ptr();
432                (Some(obj_sep_cstring), Some(array_sep_cstring), obj_sep_ptr, array_sep_ptr)
433            } else {
434                (None, None, std::ptr::null(), std::ptr::null())
435            };
436
437        cpp!(unsafe [
438            self as "xgrammar::GrammarCompiler*",
439            schema_ptr as "const char*",
440            any_whitespace as "bool",
441            has_indent as "bool",
442            indent_value as "int",
443            has_separators as "bool",
444            obj_sep_ptr as "const char*",
445            array_sep_ptr as "const char*",
446            strict_mode as "bool"
447        ] -> CompiledGrammar as "xgrammar::CompiledGrammar" {
448            std::string schema_str(schema_ptr);
449            std::optional<int> opt_indent = has_indent ? std::make_optional(indent_value) : std::nullopt;
450            std::optional<std::pair<std::string, std::string>> opt_separators;
451
452            if (has_separators) {
453                opt_separators = std::make_pair(std::string(obj_sep_ptr), std::string(array_sep_ptr));
454            } else {
455                opt_separators = std::nullopt;
456            }
457
458            return self->CompileJSONSchema(schema_str, any_whitespace, opt_indent, opt_separators, strict_mode);
459        })
460    }

538    /// Get the compiled grammar for structural tags.
539    ///
540    /// # Arguments
541    /// * `tags` - Vector of structural tag items, each containing begin, schema, and end components
542    /// * `triggers` - Vector of trigger strings
543    ///
544    /// # Returns
545    /// * A compiled grammar that can be used with GrammarMatcher
546    pub fn compile_structural_tag(
547        &self,
548        tags: &[StructuralTagItem],
549        triggers: &[String],
550    ) -> CompiledGrammar {
551        // Convert Rust data to C++ format using utility functions
552        let (_begin_cstrings, tag_begin_ptrs) =
553            Self::extract_field_to_cstring_ptrs(tags, |tag| &tag.begin);
554        let (_schema_cstrings, tag_schema_ptrs) =
555            Self::extract_field_to_cstring_ptrs(tags, |tag| &tag.schema);
556        let (_end_cstrings, tag_end_ptrs) =
557            Self::extract_field_to_cstring_ptrs(tags, |tag| &tag.end);
558        let (_trigger_cstrings, trigger_ptrs) = Self::strings_to_cstring_ptrs(triggers);
559
560        let num_tags = tags.len();
561        let num_triggers = triggers.len();
562
563        let tag_begin_ptrs_ptr = tag_begin_ptrs.as_ptr();
564        let tag_schema_ptrs_ptr = tag_schema_ptrs.as_ptr();
565        let tag_end_ptrs_ptr = tag_end_ptrs.as_ptr();
566        let trigger_ptrs_ptr = trigger_ptrs.as_ptr();
567
568        cpp!(unsafe [
569            self as "xgrammar::GrammarCompiler*",
570            tag_begin_ptrs_ptr as "const char* const*",
571            tag_schema_ptrs_ptr as "const char* const*",
572            tag_end_ptrs_ptr as "const char* const*",
573            num_tags as "size_t",
574            trigger_ptrs_ptr as "const char* const*",
575            num_triggers as "size_t"
576        ] -> CompiledGrammar as "xgrammar::CompiledGrammar" {
577            std::vector<xgrammar::StructuralTagItem> tags_vector;
578            tags_vector.reserve(num_tags);
579
580            for (size_t i = 0; i < num_tags; ++i) {
581                tags_vector.emplace_back(xgrammar::StructuralTagItem{
582                    std::string(tag_begin_ptrs_ptr[i]),
583                    std::string(tag_schema_ptrs_ptr[i]),
584                    std::string(tag_end_ptrs_ptr[i])
585                });
586            }
587
588            std::vector<std::string> triggers_vector;
589            triggers_vector.reserve(num_triggers);
590            for (size_t i = 0; i < num_triggers; ++i) {
591                triggers_vector.emplace_back(std::string(trigger_ptrs_ptr[i]));
592            }
593

618    /// Construct a BNF grammar from the json schema string.
619    pub fn from_json_schema(
620        schema: &str,
621        any_whitespace: Option<bool>,
622        indent: Option<i32>,
623        separators: Option<(String, String)>,
624        strict_mode: Option<bool>,
625        print_converted_ebnf: Option<bool>,
626    ) -> Self {
627        let schema_cstring = CString::new(schema).expect("Failed to convert schema to CString");
628        let schema_ptr = schema_cstring.as_ptr();
629        let any_whitespace = any_whitespace.unwrap_or(true);
630        let strict_mode = strict_mode.unwrap_or(true);
631        let print_converted_ebnf = print_converted_ebnf.unwrap_or(false);
632        let has_indent = indent.is_some();
633        let indent_value = indent.unwrap_or(0);
634        let has_separators = separators.is_some();
635
636        let (_obj_sep_cstring, _array_sep_cstring, obj_sep_ptr, array_sep_ptr) =
637            if let Some((obj_sep, array_sep)) = separators {
638                let obj_sep_cstring =
639                    CString::new(obj_sep).expect("Failed to convert object separator to CString");
640                let array_sep_cstring =
641                    CString::new(array_sep).expect("Failed to convert array separator to CString");
642                let obj_sep_ptr = obj_sep_cstring.as_ptr();
643                let array_sep_ptr = array_sep_cstring.as_ptr();
644                (Some(obj_sep_cstring), Some(array_sep_cstring), obj_sep_ptr, array_sep_ptr)
645            } else {
646                (None, None, std::ptr::null(), std::ptr::null())
647            };
648
649        cpp!(unsafe [
650            schema_ptr as "const char*",
651            any_whitespace as "bool",
652            has_indent as "bool",
653            indent_value as "int",
654            has_separators as "bool",
655            obj_sep_ptr as "const char*",
656            array_sep_ptr as "const char*",
657            strict_mode as "bool",
658            print_converted_ebnf as "bool"
659        ] -> Grammar as "xgrammar::Grammar" {
660            std::string schema_str(schema_ptr);
661            std::optional<int> opt_indent = has_indent ? std::make_optional(indent_value) : std::nullopt;
662            std::optional<std::pair<std::string, std::string>> opt_separators;
663
664            if (has_separators) {
665                opt_separators = std::make_pair(std::string(obj_sep_ptr), std::string(array_sep_ptr));
666            } else {
667                opt_separators = std::nullopt;
668            }
669
670            return xgrammar::Grammar::FromJSONSchema(
671                schema_str,
672                any_whitespace,
673                opt_indent,
674                opt_separators,
675                strict_mode,
676                print_converted_ebnf
677            );
678        })
679    }

695    /// Construct a grammar from a structural tag.
696    pub fn from_structural_tag(tags: &[StructuralTagItem], triggers: &[String]) -> Self {
697        let (_begin_cstrings, tag_begin_ptrs) =
698            GrammarCompiler::extract_field_to_cstring_ptrs(tags, |tag| &tag.begin);
699        let (_schema_cstrings, tag_schema_ptrs) =
700            GrammarCompiler::extract_field_to_cstring_ptrs(tags, |tag| &tag.schema);
701        let (_end_cstrings, tag_end_ptrs) =
702            GrammarCompiler::extract_field_to_cstring_ptrs(tags, |tag| &tag.end);
703        let (_trigger_cstrings, trigger_ptrs) = GrammarCompiler::strings_to_cstring_ptrs(triggers);
704
705        let num_tags = tags.len();
706        let num_triggers = triggers.len();
707
708        let tag_begin_ptrs_ptr = tag_begin_ptrs.as_ptr();
709        let tag_schema_ptrs_ptr = tag_schema_ptrs.as_ptr();
710        let tag_end_ptrs_ptr = tag_end_ptrs.as_ptr();
711        let trigger_ptrs_ptr = trigger_ptrs.as_ptr();
712
713        cpp!(unsafe [
714            tag_begin_ptrs_ptr as "const char* const*",
715            tag_schema_ptrs_ptr as "const char* const*",
716            tag_end_ptrs_ptr as "const char* const*",
717            num_tags as "size_t",
718            trigger_ptrs_ptr as "const char* const*",
719            num_triggers as "size_t"
720        ] -> Grammar as "xgrammar::Grammar" {
721            std::vector<xgrammar::StructuralTagItem> tags_vector;
722            tags_vector.reserve(num_tags);
723
724            for (size_t i = 0; i < num_tags; ++i) {
725                tags_vector.emplace_back(xgrammar::StructuralTagItem{
726                    std::string(tag_begin_ptrs_ptr[i]),
727                    std::string(tag_schema_ptrs_ptr[i]),
728                    std::string(tag_end_ptrs_ptr[i])
729                });
730            }
731
732            std::vector<std::string> triggers_vector;
733            triggers_vector.reserve(num_triggers);
734            for (size_t i = 0; i < num_triggers; ++i) {
735                triggers_vector.emplace_back(std::string(trigger_ptrs_ptr[i]));
736            }
737
738            return xgrammar::Grammar::FromStructuralTag(tags_vector, triggers_vector);
739        })
740    }

793    /// # Arguments
794    /// * `compiled_grammar` - The compiled grammar to use
795    pub fn new(compiled_grammar: &CompiledGrammar) -> Self {
796        Self::with(compiled_grammar, None, Some(true), None)
797    }

799    /// Create a GrammarMatcher from a compiled grammar.
800    /// # Arguments
801    /// * `compiled_grammar` - The compiled grammar to use
802    /// * `override_stop_tokens` - Optional list of token ids to override the default stop tokens
803    /// * `terminate_without_stop_token` - Whether to terminate the matcher without accepting a stop token.
804    /// * `max_rollback_tokens` - Deprecated. You don't need to set it and it's always unlimited (-1).
805    ///   The new Earley parser significantly reduces the number of states, so we can allow
806    ///   unlimited rollback. The maximum number of rollback tokens allowed. The rollback operation
807    ///   is useful for jump-forward decoding and speculative decoding.
808    pub fn with(
809        compiled_grammar: &CompiledGrammar,
810        override_stop_tokens: Option<&[i32]>,
811        terminate_without_stop_token: Option<bool>,
812        max_rollback_tokens: Option<i32>,
813    ) -> Self {
814        // Keep it sync with the C++ implementation:
815        // https://github.com/mlc-ai/xgrammar/blob/95bdfce011506ea95306b37d080115a2da3e369a/cpp/grammar_matcher.cc#L257
816        let terminate_without_stop_token = terminate_without_stop_token.unwrap_or(false);
817        let max_rollback_tokens = max_rollback_tokens.unwrap_or(0);
818        let override_stop_tokens_ptr =
819            override_stop_tokens.as_ref().map_or(std::ptr::null(), |v| v.as_ptr());
820        let override_stop_tokens_len = override_stop_tokens.as_ref().map_or(0, |v| v.len());
821
822        cpp!(unsafe [
823            compiled_grammar as "const xgrammar::CompiledGrammar*",
824            override_stop_tokens_ptr as "const int32_t*",
825            override_stop_tokens_len as "size_t",
826            terminate_without_stop_token as "bool",
827            max_rollback_tokens as "int"
828        ] -> GrammarMatcher as "xgrammar::GrammarMatcher" {
829            std::optional<std::vector<int32_t>> opt_override_stop_tokens;
830            if (override_stop_tokens_len > 0) {
831                opt_override_stop_tokens = std::vector<int32_t>(
832                    *override_stop_tokens_ptr,
833                    *override_stop_tokens_ptr + override_stop_tokens_len
834                );
835            } else {
836                opt_override_stop_tokens = std::nullopt;
837            }
838
839            return xgrammar::GrammarMatcher(
840                *compiled_grammar,
841                opt_override_stop_tokens,
842                terminate_without_stop_token,
843                max_rollback_tokens
844            );
845        })
846    }

848    /// Accept one token and update the state of the matcher.
849    ///
850    /// # Arguments
851    /// * `token_id` - The id of the token to accept.
852    /// * `debug_print` - If true, print debug information.
853    ///
854    /// # Returns
855    /// * Whether the token is accepted.
856    ///
857    /// # Note
858    /// Termination state.
859    ///
860    /// When the end of the root rule is reached, the matcher can only accept the stop token.
861    /// The matcher is terminated after accepting the stop token, i.e. no AcceptToken or
862    /// FindNextTokenMask operations can be performed. The termination state can be canceled
863    /// using rollback().

867            return self->AcceptToken(token_id, debug_print);
868        })
869    }
